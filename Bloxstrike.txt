-- Logic Data
-- ================================================
local r15 = {{"Head", "UpperTorso"}, {"UpperTorso", "LowerTorso"}, {"UpperTorso", "LeftUpperArm"}, {"LeftUpperArm", "LeftLowerArm"}, {"LeftLowerArm", "LeftHand"}, {"UpperTorso", "RightUpperArm"}, {"RightUpperArm", "RightLowerArm"}, {"RightLowerArm", "RightHand"}, {"LowerTorso", "LeftUpperLeg"}, {"LeftUpperLeg", "LeftLowerLeg"}, {"LeftLowerLeg", "LeftFoot"}, {"LowerTorso", "RightUpperLeg"}, {"RightUpperLeg", "RightLowerLeg"}, {"RightLowerLeg", "RightFoot"}}
local r6 = {{"Head", "Torso"}, {"Torso", "Left Arm"}, {"Torso", "Right Arm"}, {"Torso", "Left Leg"}, {"Torso", "Right Leg"}}

-- Bomb variables
local debris, char_folder
local t_start, last_bomb_state = 0, false
local tx, ty, dragging, ox, oy = 1400, 50, false, 0, 0

local function get_team(char)
    if not char or not utility.is_valid(char) then return nil end
    local curr = char
    while curr and utility.is_valid(curr) do
        if curr.Name == "Counter-Terrorists" then return "CT" end
        if curr.Name == "Terrorists" then return "T" end
        curr = curr.Parent
    end
    return nil
end

-- ================================================
-- Main Loop
-- ================================================
function on_frame()
    -- >>> PLAYER ESP <<<
    if menu.get("esp_enabled") then
        local lp = entity.get_local_player()
        local local_team = lp and get_team(lp.character) or nil

        for _, p in ipairs(entity.get_players()) do
            if p.is_local or not p.is_alive then goto skip_player end

            if menu.get("esp_teamcheck") and local_team and get_team(p.character) == local_team then goto skip_player end

            local b = p:get_bounds()
            if not b.valid then goto skip_player end

            -- Chams & Charms
            if menu.get("esp_meshchams") then draw.chams_player(p, menu.get_color("esp_meshchams"), menu.get_color("esp_meshchams2"), 0) end
            if menu.get("esp_charms") then draw.chams_player(p, menu.get_color("esp_charms"), menu.get_color("esp_charms2"), 2) end

            -- Box
            if menu.get("esp_box") then
                draw.rect_filled(b.x, b.y, b.w, b.h, menu.get_color("esp_box_fill"), 0)
                draw.box(b.x, b.y, b.w, b.h, menu.get_color("esp_box"))
            end

            -- Skeleton
            if menu.get("esp_skeleton") then
                local bones = p:get_bones_screen()
                for _, c in ipairs(p.rig_type == "R6" and r6 or r15) do
                    local n1, n2 = bones[c[1]], bones[c[2]]
                    if n1 and n2 then draw.line(n1[1], n1[2], n2[1], n2[2], menu.get_color("esp_skeleton"), 1.5) end
                end
            end

            -- Name & Distance
            if menu.get("esp_name") then
                local tw, th = draw.get_text_size(p.name, 14)
                draw.text(b.x + b.w/2 - tw/2, b.y - th - 4, p.name, menu.get_color("esp_name"), 14)
            end

            if menu.get("esp_distance") then
                local d_txt = "[" .. math.floor(p:distance_to()) .. "m]"
                local dw = draw.get_text_size(d_txt, 13)
                draw.text(b.x + b.w/2 - dw/2, b.y + b.h + 6, d_txt, menu.get_color("esp_distance"), 13)
            end

            -- Health
            if menu.get("esp_health") then draw.health_bar(b.x - 6, b.y, b.h, p.health, p.max_health) end

            -- Tracers & Look Arrow
            local hx, hy, h_vis = p:get_bone_screen("Head")
            if h_vis then
                if menu.get("esp_tracer") then
                    local sw, sh = utility.get_screen_size()
                    draw.line(sw/2, sh/2, hx, hy, menu.get_color("esp_tracer"), 1.5)
                end
                if menu.get("esp_arrow") then
                    local hp, lv = p.head_position, p.look_vector
                    local ex, ey, e_vis = draw.world_to_screen(hp.x + lv.x*6, hp.y + lv.y*6, hp.z + lv.z*6)
                    if e_vis then draw.line(hx, hy, ex, ey, menu.get_color("esp_arrow"), 2.5) end
                end
            end

            ::skip_player::
        end
    end

    -- >>> BOMB ESP <<<
    if menu.get("bomb_enabled") then
        if not utility.is_valid(debris) and utility.is_valid(game.workspace) then debris = game.workspace:find_first_child("Debris") end
        if not utility.is_valid(char_folder) and utility.is_valid(debris) then char_folder = debris:find_first_child("Character") end

        local bomb_present = utility.is_valid(char_folder)

        if bomb_present then
            -- Draggable Timer
            if menu.get("bomb_timer") then
                if not last_bomb_state then t_start = utility.get_time() end
                
                local remain = math.max(0, 40 - (utility.get_time() - t_start))
                local txt = string.format("BOMB: %.1fs", remain)
                draw.text(tx, ty, txt, menu.get_color("bomb_timer_col"), 28)

                local mx, my = utility.get_mouse_pos()
                local tw, th = draw.get_text_size(txt, 28)
                
                if input.is_key_down(0x01) then
                    if not dragging and mx > tx and mx < tx + tw and my > ty and my < ty + th then
                        dragging, ox, oy = true, mx - tx, my - ty
                    end
                else 
                    dragging = false 
                end

                if dragging then tx, ty = mx - ox, my - oy end
            end

            -- Bomb Box & Name
            for _, bomb in ipairs(char_folder:get_children()) do
                if not utility.is_valid(bomb) or not bomb:is_a("Model") then goto skip_bomb end
                local root = bomb.PrimaryPart or bomb:find_first_child("Head") or bomb:find_first_child_of_class("BasePart")
                if not utility.is_valid(root) then goto skip_bomb end

                local sx, sy, on_screen = draw.world_to_screen(root.Position.x, root.Position.y, root.Position.z)
                if not on_screen then goto skip_bomb end

                if menu.get("bomb_box") then
                    draw.box(sx - 40, sy - 80, 80, 160, menu.get_color("bomb_box_col"), 0, 0)
                end

                if menu.get("bomb_name") then
                    local nm = bomb.Name or "Bomb"
                    local tw = draw.get_text_size(nm, 18)
                    draw.text(sx - tw/2, sy - 92, nm, menu.get_color("bomb_name_col"), 18)
                end
                ::skip_bomb::
            end
        end
        last_bomb_state = bomb_present
    else
        last_bomb_state = false
    end
end
